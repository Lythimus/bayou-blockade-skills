% =============================================================================
%  bayou-filing.tex — shared LaTeX style for Bayou Blockade regulatory filings
%
%  This is a pandoc --include-in-header fragment, NOT a standalone preamble.
%  It assumes pandoc's default LaTeX template has already loaded, in order:
%      xcolor, geometry (if -V geometry given), amsmath/amssymb,
%      fontspec (xelatex/lualatex only), iftex
%  and that hyperref is loaded AFTER this file (pandoc 3.x puts
%  $hypersetup$ below $header-includes$) — which is why link colors are set
%  inside \AtBeginDocument rather than with a bare \hypersetup here.
%
%  Engine: xelatex or lualatex. Under pdflatex the font block is skipped and
%  you get the document in the class default face; everything else still works.
%
%  Usage:  see README.md in this directory.
%  Never edit the generated .tex or .pdf — markdown stays the source of truth.
% =============================================================================

\usepackage{iftex}
\usepackage{etoolbox}
\usepackage{geometry}   % idempotent: safe even if the template already loaded it
\usepackage{microtype}
\usepackage[shortlabels]{enumitem}
\usepackage{booktabs}
\usepackage{array}
\usepackage{titlesec}
\usepackage{fancyhdr}
\usepackage{lastpage}
\usepackage{framed}
\usepackage{needspace}

% ---- CAMPAIGN IDENTIFIERS ---------------------------------------------------
%  Set these in a small per-campaign file passed as an EARLIER
%  --include-in-header than this one (see README.md):
%
%    \def\filingkind{Public Comment and Request for Public Hearing}
%    \def\filingid{AI 41475 · Permit 7777-00936-00 · Activity PER20250001}
%    \def\filingagency{Louisiana Department of Environmental Quality}
%    \def\filingshort{ExxonMobil Pipeline — Statewide Flaring}
%    \def\filingauthorline{Jane Doe · Anytown, Example Parish}
%
%  Do NOT put these in the markdown's YAML `header-includes` instead.
%  --include-in-header sets the header-includes *variable*, and a pandoc
%  variable shadows the metadata field of the same name, so YAML overrides are
%  silently dropped the moment this file is passed with -H. Verified against
%  pandoc 3.9; the failure is silent, which is what makes it worth a warning.
%
%  Multiple -H files are concatenated in the order given, so the campaign file
%  runs first and its \def wins over the \providecommand fallbacks below.
%  Do not change these to \newcommand.
\providecommand{\filingkind}{Public Comment}
\providecommand{\filingid}{}
\providecommand{\filingagency}{}
\providecommand{\filingshort}{}
\providecommand{\filingauthorline}{}

% ---- PAGE GEOMETRY ----------------------------------------------------------
\geometry{
  letterpaper,
  top=1.05in, bottom=1.05in, left=1.15in, right=1.15in,
  headheight=15pt, footskip=28pt
}

% ---- COLORS -----------------------------------------------------------------
\definecolor{bayouBlue}{RGB}{20,40,100}
\definecolor{bayouRed}{RGB}{160,20,20}
\definecolor{bayouGrayLt}{RGB}{240,240,240}
\definecolor{bayouGrayMid}{RGB}{150,150,150}
\definecolor{bayouGrayDk}{RGB}{90,90,90}

% ---- FONTS ------------------------------------------------------------------
%  Loaded by filename, not by family name: TeX Gyre Pagella ships with TeX Live
%  but is not registered with fontconfig on macOS, so \setmainfont{TeX Gyre
%  Pagella} fails while the file-based form below resolves through kpathsea.
%  Lining (not old-style) figures are deliberate — permit and AI numbers must
%  read unambiguously.
\ifPDFTeX\else
  \setmainfont{texgyrepagella}[
    Extension      = .otf,
    UprightFont    = *-regular,
    BoldFont       = *-bold,
    ItalicFont     = *-italic,
    BoldItalicFont = *-bolditalic,
    Ligatures      = TeX
  ]
\fi

% ---- BODY SETTINGS ----------------------------------------------------------
\setlength{\parindent}{0pt}
\setlength{\parskip}{7pt plus 2pt minus 1pt}
\linespread{1.04}
\setlist{topsep=4pt, itemsep=3pt, parsep=2pt, leftmargin=*}
\raggedbottom

% Discourage orphaned headings and stranded single lines.
\clubpenalty=10000
\widowpenalty=10000
\brokenpenalty=10000

% ---- SECTION HEADINGS -------------------------------------------------------
%  pandoc sets secnumdepth to -\maxdimen unless --number-sections is passed,
%  and the comment template hand-numbers its parts ("## IV. Notice defects"),
%  so the label argument is intentionally empty — no LaTeX-generated numbers.
\titleformat{\section}
  {\needspace{3\baselineskip}\large\bfseries\color{bayouBlue}}
  {}{0pt}{}[\vspace{2pt}{\color{bayouGrayMid}\titlerule}\vspace{2pt}]
\titleformat{\subsection}
  {\needspace{2.5\baselineskip}\normalsize\bfseries\color{bayouBlue}}
  {}{0pt}{}
\titleformat{\subsubsection}
  {\needspace{2\baselineskip}\normalsize\itshape\color{bayouGrayDk}}
  {}{0pt}{}

\titlespacing*{\section}{0pt}{20pt plus 4pt minus 2pt}{7pt plus 2pt}
\titlespacing*{\subsection}{0pt}{13pt plus 3pt minus 1pt}{4pt plus 1pt}
\titlespacing*{\subsubsection}{0pt}{11pt plus 2pt}{3pt}

% ---- RUNNING HEAD / FOOT ----------------------------------------------------
%  The footer carries the permit identifiers on EVERY page. Agencies scan
%  filings into document-management systems (LDEQ EDMS) where pages are
%  separated from their cover; an unlabelled page 14 is an orphan.
\pagestyle{fancy}
\fancyhf{}
\renewcommand{\headrulewidth}{0.4pt}
\renewcommand{\footrulewidth}{0.4pt}
\renewcommand{\headrule}{\hbox to\headwidth{\color{bayouGrayMid}\leaders\hrule height \headrulewidth\hfill}}
\renewcommand{\footrule}{\hbox to\headwidth{\color{bayouGrayMid}\leaders\hrule height \footrulewidth\hfill}}
\fancyhead[L]{\footnotesize\color{bayouBlue}\filingshort}
\fancyhead[R]{\footnotesize\color{bayouBlue}\filingkind}
\fancyfoot[L]{\scriptsize\color{bayouGrayDk}\filingid}
\fancyfoot[R]{\scriptsize\color{bayouGrayDk}Page \thepage\ of \pageref{LastPage}}

% First page: identifiers in the footer, but no running head competing with
% the letterhead block.
\fancypagestyle{bayoufirst}{%
  \fancyhf{}%
  \renewcommand{\headrulewidth}{0pt}%
  \renewcommand{\footrulewidth}{0.4pt}%
  \fancyfoot[L]{\scriptsize\color{bayouGrayDk}\filingid}%
  \fancyfoot[R]{\scriptsize\color{bayouGrayDk}Page \thepage\ of \pageref{LastPage}}%
}

% ---- LETTERHEAD -------------------------------------------------------------
%  Replaces pandoc's \maketitle. Driven by the markdown's YAML `title:` and
%  `date:` plus the \filing* macros above. pandoc only emits \maketitle when
%  `title:` is present, so the front matter is what switches this on.
\makeatletter
\renewcommand{\maketitle}{%
  \thispagestyle{bayoufirst}%
  \begingroup
  \setlength{\parskip}{0pt}%
  {\color{bayouGrayMid}\hrule height 1.5pt}%
  \vspace{9pt}%
  {\footnotesize\bfseries\color{bayouRed}\MakeUppercase{\filingkind}\par}%
  \vspace{7pt}%
  {\LARGE\bfseries\color{bayouBlue}\@title\par}%
  \vspace{9pt}%
  {\color{bayouGrayMid}\hrule height 0.6pt}%
  \vspace{7pt}%
  % etoolbox's \ifdefempty, not \ifx...\@empty: \providecommand defines a \long
  % macro, \@empty is not \long, and \ifx compares those as unequal — so the
  % \ifx form printed empty "Submitted to" lines for unset fields.
  {\small
    \ifdefempty{\filingid}{}{\textbf{\filingid}\par\vspace{2pt}}%
    \ifdefempty{\filingagency}{}{Submitted to \filingagency\par\vspace{2pt}}%
    \ifdefempty{\@date}{}{\@date\par}%
    \ifdefempty{\filingauthorline}{}{\vspace{2pt}\filingauthorline\par}%
  }%
  \vspace{9pt}%
  {\color{bayouGrayMid}\hrule height 1.5pt}%
  \endgroup
  \vspace{16pt}%
}
\makeatother

% ---- CALLOUT ENVIRONMENTS ---------------------------------------------------
%  Reached from markdown via fenced divs plus bayou-filing.lua:
%      ::: comment      → filingcomment
%      ::: recordquote  → filingrecordquote
%      ::: alert        → filingalert
%  All three are built on `framed` so they break across pages — a comment that
%  cannot split is a comment that pushes a half-empty page into the filing.

% The requested permit condition. This is the load-bearing element of the
% document: LDEQ's Public Comments Response Summary is built by walking the
% numbered comments, so each one has to be findable by eye in a 40-page PDF.
\newenvironment{filingcomment}{%
  \def\FrameCommand{{\color{bayouBlue}\vrule width 3pt}\hspace{10pt}}%
  \vspace{4pt}%
  \MakeFramed{\advance\hsize-\width\FrameRestore}%
  \setlength{\parskip}{5pt}%
}{\endMakeFramed\vspace{4pt}}

% Verbatim material quoted out of the agency's own record.
\newenvironment{filingrecordquote}{%
  \def\FrameCommand{{\color{bayouGrayMid}\vrule width 1pt}\hspace{10pt}}%
  \vspace{4pt}%
  \MakeFramed{\advance\hsize-\width\FrameRestore}%
  \small\itshape
  \setlength{\parskip}{5pt}%
}{\endMakeFramed\vspace{4pt}}

% Shaded emphasis box, matching the \alertbox used in the hand-written filings.
\newenvironment{filingalert}{%
  \def\FrameCommand{\colorbox{bayouGrayLt}}%
  \vspace{6pt}%
  \MakeFramed{\advance\hsize-\width\FrameRestore}%
  \small
  \setlength{\parskip}{5pt}%
}{\endMakeFramed\vspace{6pt}}

% pandoc emits plain blockquotes as `quote`; keep it visually distinct from
% the three callouts above so an ordinary quotation is never mistaken for a
% requested condition.
\renewenvironment{quote}
  {\list{}{\leftmargin=2em\rightmargin=2em}\item\relax\small}
  {\endlist}

% ---- TABLES -----------------------------------------------------------------
\renewcommand{\arraystretch}{1.15}
\setlength{\tabcolsep}{6pt}

% ---- LINKS ------------------------------------------------------------------
%  Deferred: hyperref is loaded after this fragment, so this must run later.
%  Printed filings are read on paper — links are colored for the screen copy
%  but never boxed, and `breaklinks` keeps long EDMS URLs inside the margin.
\AtBeginDocument{%
  \hypersetup{
    colorlinks = true,
    linkcolor  = bayouBlue,
    urlcolor   = bayouBlue,
    citecolor  = bayouBlue,
    breaklinks = true,
    pdfborder  = {0 0 0}
  }%
}
